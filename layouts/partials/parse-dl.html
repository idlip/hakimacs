{{- /* parse-dl: parses one dt/dd pair from org-rendered description list HTML. Input: dict "dt"
...
"dd"
...
Returns dict: url — href from the
link, empty string if the term isn't linked name — term text, tags stripped, HTML-unescaped desc — first segment before " | " parts — slice of remaining " | "-split segments (trimmed), for caller to parse */ -}} {{- $dt := .dt -}} {{- $dd := .dd -}} {{- $url := "" -}} {{- if findRE `]+)"?.*` "$1" -}} {{- end -}} {{- $name := $dt | replaceRE `(?s).*?]+>(.*?).*` "$1" | htmlUnescape | replaceRE `<[^>]+>` "" | strings.TrimSpace -}} {{- $raw := $dd | replaceRE `(?s)
\s*(.*?)\s*
` "$1" -}} {{- $segs := split $raw " | " -}} {{- return dict "url" $url "name" $name "desc" (index $segs 0) "parts" (after 1 $segs) -}}
