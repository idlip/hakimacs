{{- /* scheme-swatch — input: a params.colorSchemes entry. Returns the menu swatch style "--p-bg:..;--p-fg:..;--p-accent:..". Derives from the base16 yaml (base00/base05/base0D) when the entry has a `dark` key; otherwise uses the entry's explicit bg/fg/accent (haki, Modus). */ -}} {{- $e := . -}} {{- $bg := $e.bg -}}{{- $fg := $e.fg -}}{{- $accent := $e.accent -}} {{- with $e.dark -}} {{- with (index site.Data.schemes.base16 .) -}} {{- $bg = .palette.base00 -}} {{- $fg = .palette.base05 -}} {{- $accent = or (index .palette "base0D") (index .palette "base0d") -}} {{- end -}} {{- end -}} {{- return (printf "--p-bg:%s;--p-fg:%s;--p-accent:%s" $bg $fg $accent) -}}
