{{- /* References + See Also, merged into one partial. Collects in-content links (external http(s) and internal /page links, deduped, footnote/TOC anchors and mailto: excluded) as a numbered References list. A shared link budget (Site.Params.linkBudget, default 7) keeps the trailing link block from crowding: See Also related posts only fill UP TO the budget when references are few, and are hidden entirely once references >= budget. Related posts already cited in-content are excluded (no duplicates). Usage: {{ partial "references.html" . }} */ -}} {{- $budget := .Site.Params.linkBudget | default 7 -}} {{- /* 1. Collect references from post content. */ -}} {{- $matches := findRESubmatch `(?s)]+href="([^"]+)"[^>]*>(.*?)` .Content -}} {{- $seen := slice -}} {{- $refs := slice -}} {{- $refURLs := slice -}} {{- range $matches -}} {{- $href := index . 1 -}} {{- $text := index . 2 | plainify | htmlUnescape | strings.TrimSpace -}} {{- $isExt := or (hasPrefix $href "http://") (hasPrefix $href "https://") -}} {{- $isInt := hasPrefix $href "/" -}} {{- if and (or $isExt $isInt) (not (in $seen $href)) -}} {{- $seen = $seen | append $href -}} {{- $refs = $refs | append (dict "href" $href "text" $text "internal" $isInt) -}} {{- if $isInt -}} {{- $refURLs = $refURLs | append (strings.TrimSuffix "/" $href) -}} {{- end -}} {{- end -}} {{- end -}} {{- $n := len $refs -}} {{- /* 2. Build See Also fill only while under budget. */ -}} {{- $related := slice -}} {{- if lt $n $budget -}} {{- $tags := .Params.tags -}} {{- $tag1 := index $tags 0 -}} {{- $tag2 := index $tags 1 -}} {{- $frefs := .Params.refs -}} {{- $ref1 := index $frefs 0 -}} {{- $ref2 := index $frefs 1 -}} {{- if or $tag1 $tag2 $ref1 $ref2 -}} {{- $current := . -}} {{- range .Site.RegularPages -}} {{- $tagMatch := or (and $tag1 (in .Params.tags $tag1)) (and $tag2 (in .Params.tags $tag2)) -}} {{- $refMatch := or (and $ref1 (in .Params.refs $ref1)) (and $ref2 (in .Params.refs $ref2)) -}} {{- $alreadyRef := in $refURLs (strings.TrimSuffix "/" .RelPermalink) -}} {{- if and (ne . $current) (or $tagMatch $refMatch) (not $alreadyRef) -}} {{- $related = $related | append . -}} {{- end -}} {{- end -}} {{- $related = first (sub $budget $n) $related -}} {{- end -}} {{- end -}} {{- /* 3. Render. References first, then See Also. */ -}} {{- with $refs -}}
